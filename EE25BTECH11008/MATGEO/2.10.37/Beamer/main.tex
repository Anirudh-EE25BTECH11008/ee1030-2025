%----------------- Problem -------------------
\begin{frame}{Problem Statement}
If $\vec{a}$, $\vec{b}$ and $\vec{c}$ are three non-coplanar vectors, then find the value of  
\[
(\vec{a} + \vec{b} + \vec{c}) \cdot [(\vec{a} + \vec{b}) \times (\vec{a} + \vec{c})].
\]
\end{frame}

%----------------- Solution -------------------
\begin{frame}{Solution}
We know that the scalar triple product can be written as a determinant:
\[
[\vec{p}\ \vec{q}\ \vec{r}] \;=\; \vec{p}\cdot(\vec{q}\times\vec{r}) \;=\; \mydet{\vec{p} & \vec{q} & \vec{r}}.
\]

\begin{align}
(\vec{a} + \vec{b} + \vec{c}) \cdot [(\vec{a} + \vec{b}) \times (\vec{a} + \vec{c})]
&= \mydet{\vec{a}+\vec{b}+\vec{c} & \vec{a}+\vec{b} & \vec{a}+\vec{c}}.
\end{align}
\end{frame}

%----------------- Solution (cont) -------------------
\begin{frame}{Solution (cont..)}
\begin{align}
\mydet{\vec{a}+\vec{b}+\vec{c} & \vec{a}+\vec{b} & \vec{a}+\vec{c}}
&= \mydet{\vec{a} & \vec{b} & \vec{c}}
\;\mydet{
1 & 1 & 1\\[4pt]
1 & 1 & 0\\[4pt]
1 & 0 & 1
}.
\end{align}

\begin{align}
\mydet{
1 & 1 & 1\\[4pt]
1 & 1 & 0\\[4pt]
1 & 0 & 1
} &= -1.
\end{align}
\end{frame}

%----------------- Solution (cont) -------------------
\begin{frame}{Solution (cont..)}
Therefore
\begin{align}
(\vec{a} + \vec{b} + \vec{c}) \cdot [(\vec{a} + \vec{b}) \times (\vec{a} + \vec{c})]
&= \mydet{\vec{a} & \vec{b} & \vec{c}} \cdot (-1) \\[4pt]
&= -[\vec{a}\ \vec{b}\ \vec{c}].
\end{align}

\[
\boxed{(\vec{a} + \vec{b} + \vec{c}) \cdot [(\vec{a} + \vec{b}) \times (\vec{a} + \vec{c})] = -[\vec{a}\ \vec{b}\ \vec{c}]}
\]
\end{frame}
